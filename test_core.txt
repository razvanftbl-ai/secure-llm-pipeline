import seclib

key = seclib.generate_key()
print("cheie:", key.hex())

# AES-GCM
blob = seclib.encrypt(key, "salut, sunt Andrei".encode(), b"ctx")
print("criptat:", blob.hex())
assert seclib.decrypt(key, blob, b"ctx") == "salut, sunt Andrei".encode()

# integritate: modificam un byte
bad = bytearray(blob); bad[20] ^= 1
try:
    seclib.decrypt(key, bytes(bad), b"ctx")
    print("EROARE: tamper nedetectat")
except ValueError:
    print("OK: tamper detectat")

# Shamir 3-din-5
shares = seclib.split_secret(key, 3, 5)
assert seclib.combine_shares(shares[:3]) == key
assert seclib.combine_shares([shares[0], shares[2], shares[4]]) == key
assert seclib.combine_shares(shares[:2]) != key   # 2 share-uri nu ajung
print("OK: Shamir 3-din-5 functioneaza")
